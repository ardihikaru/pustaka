Block diagrams

Model signal paths, transfer blocks, terminals, summing junctions, takeoffs, infrastructure icons, and annotations in editable or fixed read-only form.

Live block-diagram workbench

The boxed sample starts editable and renders the complete upstream control-system model.

Live block-diagram viewer

The unboxed viewer is read-only from creation and contains no palette or sidebar.

Fixed and read-only contract

Authored coordinates preserve the control-system geometry. Global readOnly: true prevents node movement, label editing, new connections, reshape, and history mutation while keeping viewport navigation and export enabled.

javascript
import sisflow from './vendor/sisflow/index.js';
import { SISMAYA_LIGHT } from './vendor/sisflow/theme/themes.js';
import { defineBlockDiagramShapes } from './vendor/pages/block-diagram-setup.js';

sisflow.themes.register('docs-light', SISMAYA_LIGHT);
const container = document.querySelector('#diagram'); // position:relative; height:520px
const model = {
  sisflow: 1,
  nodes: [
    { id: 'input', shape: 'bd-input', x: 40, y: 140, data: { label: 'Reference' } },
    { id: 'sum', shape: 'bd-summing', x: 190, y: 130, data: { label: '+/-' } },
    { id: 'controller', shape: 'bd-block', x: 330, y: 116, data: { label: 'Controller\nC(s)' } },
    { id: 'plant', shape: 'bd-block', x: 560, y: 116, data: { label: 'Plant\nG(s)' } },
    { id: 'output', shape: 'bd-output', x: 790, y: 140, data: { label: 'Output' } }
  ],
  edges: [
    { id: 'e1', from: 'input', to: 'sum' }, { id: 'e2', from: 'sum', to: 'controller' },
    { id: 'e3', from: 'controller', to: 'plant', data: { label: 'u' } },
    { id: 'e4', from: 'plant', to: 'output', data: { label: 'y' } }
  ]
};
const flow = sisflow.create(container, {
  renderer: 'canvas', theme: 'docs-light', readOnly: true,
  grid: { size: 8, snap: true }, edges: { route: 'ortho', corner: 5, jumpOver: true }
});
defineBlockDiagramShapes(flow);
flow.graph.load(model);
flow.commands.zoomToFit(24);
const toggleFullscreen = (target) => document.fullscreenElement
  ? document.exitFullscreen() : target.requestFullscreen();
document.querySelector('#canvas-full').onclick = () => toggleFullscreen(container);
document.querySelector('#viewer-full').onclick = () => toggleFullscreen(document.querySelector('#viewer'));
document.addEventListener('fullscreenchange', () => { flow.resize(); flow.commands.zoomToFit(24); });
document.querySelector('#export').onclick = () => saveSVG(flow.export.toSVG({ padding: 16 }));
addEventListener('beforeunload', () => flow.dispose());
function saveSVG(text) {
  const url = URL.createObjectURL(new Blob([text], { type: 'image/svg+xml' }));
  const a = Object.assign(document.createElement('a'), { href: url, download: 'block-diagram.svg' });
  a.click(); URL.revokeObjectURL(url);
}

Canvas and viewer fullscreen

Use the canvas as the Fullscreen API target to remove all host chrome, or use the viewer wrapper to retain controls. An iframe host must explicitly allow fullscreen.

Rebuild and disposal

Dispose before rebuilding theme or read-only state, then define shapes, reload the unchanged canonical model, and fit the replacement surface.
